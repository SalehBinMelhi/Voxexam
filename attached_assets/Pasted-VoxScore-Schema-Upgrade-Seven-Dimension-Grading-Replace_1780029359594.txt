=== STEP 1 — ADD NEW TYPES IN shared/schema.ts ===

Add the following types:

// VoxScore dimension IDs
type VoxDimension = 'D1' | 'D2' | 'D3' | 'D4' | 'D5' | 'D6' | 'D7'

// Band score per dimension: 1=Inadequate, 2=Limited, 3=Developing, 4=Proficient, 5=Exemplary
type VoxBand = 1 | 2 | 3 | 4 | 5

// Per-dimension score profile
type VoxDimensionScore = {
  dimension: VoxDimension
  band: VoxBand
  weightedScore: number       // band × dimension weight × 100
  evidence: string            // transcript excerpt or benchmark concept note
  conceptsPresent: string[]   // benchmark concepts found in answer
  conceptsMissing: string[]   // benchmark concepts not found
  conceptsIncorrect: string[] // benchmark concepts present but wrong
}

// Full VoxScore profile
type VoxScoreProfile = {
  dimensions: VoxDimensionScore[]
  totalScore: number          // sum of all weightedScores, 0-100
  passFail: 'pass' | 'fail'   // pass threshold is 60
  confidenceLevel: 'high' | 'medium' | 'low'
  asrQualityFlag: boolean     // true if ASR confidence was below threshold
  languageDetected: 'english' | 'arabic' | 'mixed'
}

// Professor decision on a submission
type ProfessorDecision = 'accepted' | 'adjusted' | 'overridden'
